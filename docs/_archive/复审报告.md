# 报告导出功能修复方案

## 一、问题概述

当前报告预览（HTML）和报告导出（DOCX）存在以下问题：

### 1.1 HTTP 422 错误

前端调用 `POST /api/v1/documents/export` 返回 422 Unprocessable Content。

### 1.2 导出功能无实际效果

即使 422 修复，导出流程也只是返回 mock 数据，没有真正生成 DOCX 文件。

### 1.3 版本不同步

前端写死 `version_id: 'v1'`，但报告可能经过多次修订。

---

## 二、Root Cause 分析

### 2.1 422 错误的直接原因

**前端发送**（`web/src/lib/api.ts:314-317`）：

```typescript
async exportDocument(taskId: string, versionId: string, format: string): Promise<Blob> {
    const { data } = await this.client.post('/api/v1/documents/export', 
      { task_id: taskId, version_id: versionId, format },  // 3 个字段
      { responseType: 'blob' }
    );
    return data;
}
```

**后端期望**（`src/api/document_api.py:122-128`）：

```python
@dataclass
class ExportDocumentRequest:
    task_id: str
    version_id: str
    format: str
    export_path: str   # ← 第 4 个必填字段，前端未发送
```

前端发送 3 个字段，但 `@dataclass` 要求 4 个必填字段。`export_path` 缺失 → FastAPI 无法构造 `@dataclass` → 返回 422。

### 2.2 导出功能空转的原因

`main.py:600` 创建 `DocumentAPI` 时未传入任何 manager：

```python
document_api = DocumentAPI()
```

所有依赖注入字段都是 `None`，包括 `_export_manager`。因此 `_export_document` 走 mock 分支：

```python
# Mock implementation
return {
    "status": "success",
    "export_id": f"export_{...}",
    "export_path": request.export_path
}
```

后端并没有真正执行 HTML → DOCX 的转换。

### 2.3 下载端点不可用

`main.py:98-110` 的下载端点：

```python
@app.get("/api/v1/download/{task_id}")
async def download_document(task_id: str):
    task_dir = Path("data") / task_id
    docs = sorted(task_dir.glob("*.docx")) + sorted(task_dir.glob("*.html"))
    if not docs:
        raise HTTPException(status_code=404, detail="Document not found")
```

- `data/{task_id}/` 目录可能不存在（task_id 是会话 ID 如 `ses_b6bef8b5`）
- 即使存在，里面也没有 `.docx` 文件

## 三、修复方案

### 3.1 修改 `ExportDocumentRequest` 模型

**文件**: `src/api/document_api.py`

```python
@dataclass
class ExportDocumentRequest:
    task_id: str
    version_id: str
    format: str
    export_path: Optional[str] = None   # ← 改为可选

    def validate(self) -> Optional[str]:
        if not self.task_id or not TASK_ID_PATTERN.match(self.task_id):
            return f"Invalid task_id: {self.task_id}"
        # version_id: 接受 'latest' 或 v\d+ 格式
        if self.version_id != 'latest' and not VERSION_ID_PATTERN.match(self.version_id):
            return f"Invalid version_id: {self.version_id}"
        if self.format not in VALID_OUTPUT_FORMATS:
            return f"Invalid format: {self.format}"
        # export_path 为空时跳过路径校验
        if self.export_path:
            for pattern in DANGEROUS_PATH_PATTERNS:
                if pattern.lower() in self.export_path.lower():
                    return f"Unsafe export path detected"
        return None
```

同时修改 `export_document` 方法中 `_is_safe_path` 调用处，增加 None 守卫：

```python
# document_api.py:486
if request.export_path and not self._is_safe_path(request.export_path):
    logger.warning(f"Unsafe export path: {request.export_path}")
    return {"status": "failed", "error": "Unsafe export path"}
```

### 3.2 重写导出逻辑

**文件**: `src/api/document_api.py`

```python
async def _export_document(self, request: ExportDocumentRequest) -> Dict[str, Any]:
    """将最新的 HTML 预览转换为 DOCX"""

    task_id = request.task_id
    # 1. 读取最新 HTML 预览
    preview_path = Path("data/previews") / f"{task_id}.html"
    if not preview_path.exists():
        return {"status": "failed", "error": "Preview not found. Generate preview first."}

    html_content = preview_path.read_text(encoding="utf-8")

    # 2. 确保输出目录存在
    output_dir = Path("data") / task_id
    output_dir.mkdir(parents=True, exist_ok=True)

    output_path = str(output_dir / f"{task_id}_report.docx")

    # 3. 使用 HTMLToWordConverter 转换
    from src.converters.html_to_word import HTMLToWordConverter
    converter = HTMLToWordConverter()
    result = converter.convert(html=html_content, output_path=output_path)

    if not result.success:
        return {"status": "failed", "error": result.error}

    # 4. 返回成功 + 下载 URL
    return {
        "status": "success",
        "download_url": f"/api/v1/download/{task_id}",
        "file_name": f"{task_id}_report.docx",
        "file_size": result.file_size,
    }
```

### 3.3 修改下载端点

**文件**: `src/api/main.py`

当前端点按 `data/{task_id}/` 目录查找，但 task_id 是会话 ID（如 `ses_b6bef8b5`），需修改为正确的路径：

```python
@app.get("/api/v1/download/{task_id}")
async def download_document(task_id: str):
    # 优先查找 data/{task_id}/ 目录下的 .docx（导出后的文件）
    task_dir = Path("data") / task_id
    if task_dir.is_dir():
        docs = sorted(task_dir.glob("*.docx"))
        if docs:
            return _FastAPIFileResponse(
                str(docs[-1]),
                filename=docs[-1].name,
                media_type="application/vnd.openxmlformats-officedocument.wordprocessingml.document",
            )

    # 回退：查找 data/previews/ 下的 .html 预览
    preview_path = Path("data/previews") / f"{task_id}.html"
    if preview_path.is_file():
        return _FastAPIFileResponse(
            str(preview_path),
            filename=f"{task_id}.html",
            media_type="text/html",     # ← 修正：HTML 回退必须用 text/html
        )

    raise HTTPException(status_code=404, detail="Document not found")
```

### 3.5 修改前端组件流程

**文件**: `web/src/components/preview/DocumentPreview.tsx`

```typescript
const handleFinalize = async () => {
    if (!taskId) return;
    setFinalizing(true);
    setError(null);
    try {
        const res = await api.exportDocument(taskId, 'latest', 'docx');
        if (res.status === 'success') {
            setFinalized(true);
            setDownloadUrl(res.download_url!);
        } else {
            setError(res.error || 'Finalize failed');
        }
    } catch (e: any) {
        setError(e?.message || 'Finalize failed');
    } finally {
        setFinalizing(false);
    }
};

const handleDownload = useCallback(async () => {
    if (!taskId || !downloadUrl) return;
    try {
        const response = await fetch(downloadUrl);
        if (!response.ok) {
            setError('Download failed: file not found');
            return;
        }
        const blob = await response.blob();
        const url = window.URL.createObjectURL(blob);
        const a = document.createElement('a');
        a.href = url;
        a.download = 'report.docx';
        a.click();
        window.URL.revokeObjectURL(url);
    } catch (e: any) {
        setError(e?.message || 'Download failed');
    }
}, [taskId, downloadUrl]);
```

## 五、注意事项

1. **python-docx 依赖**：确保运行环境已安装 `python-docx`，否则 `HTMLToWordConverter` 无法工作
2. **HTML 预览必须先生成**：导出流程依赖 `data/previews/{task_id}.html` 存在，如果预览未生成需先调 preview 接口
3. **中文字体**：`HTMLToWordConverter` 默认使用 `Microsoft YaHei`，Linux 环境可能需要调整字体配置
